// Остаток от деления

#include <iostream> //Подключение библиотеки ввода-вывода

int main() {		//Точка входа
	setlocale(LC_ALL, "RU"); //Функция для отображения кириллицы

	///Объявление переменных
	int totalDays = 0; //Входное число дней
	int years = 0; //Количество лет
	int months = 0; //Количество месяцев
	int days = 0; //Количество дней
	int remainingDays = 0; //Количество оставшихся дней

	//Ввод количества дней
	std::cout << "Введите количество дней: ";
	std::cin >> totalDays;
	while (std::cin.fail() || totalDays < 0) { //Цикл с проверкой на корректный ввод
		std::cout << "Ошибка ввода! Введите целое число: "; //Предупреждение об ошибке
		std::cin.clear();	//Сброс флага ошибки
		std::cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n');  //Выбрасываем содержание из буфера
		std::cin >> totalDays;
	}

	//Вычисления количества лет, месяцев, дней
	years = totalDays / 365; //Вычисление количества лет

	remainingDays = totalDays - years * 365; //Вычиление оставшихся дней

	months = remainingDays / 30; //Вычисление количества месяцев

	days = remainingDays - months * 30; //Вычисление количества дней

	//Вывод результата
	std::cout << "Количество лет: " << years << '\n';
	std::cout << "Количество месяцев: " << months << '\n';
	std::cout << "Количество дней: " << days << '\n';


	return 0; //Успешное завершение
}
